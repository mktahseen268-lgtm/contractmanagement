#   dev/CI     SoftHSM2      apt-get install softhsm2   -> /usr/lib/softhsm/libsofthsm2.so
#   production Thales Luna   /usr/safenet/lunaclient/lib/libCryptoki2_64.so
#              Utimaco       /opt/utimaco/lib/libcs_pkcs11_R3.so
#              Entrust nShield /opt/nfast/toolkits/pkcs11/libcknfast.so
#
# See docs/PKI-ARCHITECTURE.md for the key-ceremony and slot-configuration notes.
